function walk(o, id, out) {
  if (Array.isArray(o)) { o.forEach((x, i) => walk(x, `${id}[${i}]`, out)); return }
  if (o && typeof o === 'object') { const nid = o.id ? `${id}:${o.id}` : id; for (const [k, v] of Object.entries(o)) { if (typeof v === 'string' && /^(text|say|request|utterance|input|prompt|message)$/.test(k)) out.push([`${nid}.${k}`, v]); else walk(v, nid, out) } }
}
