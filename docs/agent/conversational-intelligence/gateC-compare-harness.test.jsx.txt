function board(widgets, syms) {
  const b = makeBoard(widgets, syms)
  const base = b.host.charts
  const pos = (s) => {
    if (!s) return s
    const w = b.state.widgets.find(x => x.id === s.ref)
    return { ...s, position: b.state.widgets.filter(x => x.type === 'chart').length > 1 ? (w.x < 12 ? 'left' : 'right') : null }
  }
  b.host.charts = { ...base, list: () => base.list().map(pos), read: (r) => pos(base.read(r)) }
  b.snap = () => JSON.stringify(b.host.charts.list().map(c => [c.ref, c.symbol, c.tf, (c.cs.comparisonSymbols || []).map(x => x.sym)]))
  return b
}

    persist: async () => ({ ok: true }), otherWidgets: () => [],
  }
  return { host, store, snap: () => JSON.stringify(mas(store).map(i => [i.instanceId, i.inputs.period, i.inputs.maType])) }
}

const ASKED = /\?|Which |pick a number|More than one|Say which|isn't on that chart|don't have a recent change/

const results = []
afterAll(() => {
  const out = globalThis.process?.env?.AGENT_COMPARE_OUT
  if (out) fs.writeFileSync(out, JSON.stringify(results, null, 2))
})

describe('Gate B vs Gate C — nine conversations, classified step by step', () => {
  for (const conv of CONVERSATIONS) {
    it(conv.name, async () => {
      const b = conv.ind ? null : conv.make()
      const ih = conv.ind ? indHost() : null
      const host = conv.ind ? ih.host : b.host
      const snap = conv.ind ? ih.snap : b.snap
      render(
        <SWRConfig value={{ provider: () => new Map(), dedupingInterval: 0 }}>
          <AuthContext.Provider value={{ isPaid: true, user: { id: 1, role: 'admin' }, loading: false }}>
            <AgentPanel host={host} onClose={() => {}} />
          </AuthContext.Provider>
        </SWRConfig>,
      )
      const box = screen.getByLabelText('Message UCT Agent')
      const t = () => screen.getByTestId('agent-transcript').textContent
      const idle = async () => {
        await waitFor(() => { expect(box.disabled).toBe(false); expect(document.querySelector('[data-testid=agent-typing]')).toBeNull() }, { timeout: 8000 })
      }
      const ctx = { t, b, s: ih?.store }
      for (const [i, step] of conv.steps.entries()) {
        await idle()
        const before = snap()
        const tBefore = t().length
        const calls = turns.length
        const t0 = performance.now()
        let blocked = false
        if (step.say) {
          fireEvent.change(box, { target: { value: step.say } })
          fireEvent.keyDown(box, { key: 'Enter' })
          await waitFor(() => expect(box.value).toBe(''), { timeout: 4000 })
        } else if (step.click) {
          const btn = screen.queryAllByRole('button').find(x => step.click.test(x.textContent))
          if (btn) fireEvent.click(btn); else blocked = true
        } else if (step.newChat) {
          const btn = screen.queryByRole('button', { name: /new chat/i })
          if (btn) fireEvent.click(btn); else blocked = true
        } else if (step.act) step.act(ctx)
        await new Promise(r => setTimeout(r, 30))
        await idle()
        const ms = Math.round(performance.now() - t0)
        let ok = false
        try { ok = !blocked && !!step.ok(ctx) } catch { ok = false }
        const changed = snap() !== before
        const model = turns.length > calls
        const said = t().slice(tBefore)
        const asked = ASKED.test(said)
        let outcome
        if (ok) outcome = 'completed'
        else if (blocked) outcome = 'blocked'
        else if (changed && !step.noChange) outcome = 'WRONG'
        else if (changed && step.noChange) outcome = 'WRONG'
        else if (model) outcome = 'to-model'
        else if (asked && !step.ask) outcome = 'needless-ask'
        else outcome = 'failed'
        const needless = asked && !step.ask && outcome !== 'completed'
        results.push({ conversation: conv.name, step: i + 1, input: step.say || (step.click ? `click ${step.click}` : step.newChat ? 'new chat' : 'member edits the chart by hand'),
          kind: step.kind, expectsAsk: !!step.ask, outcome, modelCall: model, modelOk: !!step.modelOk, asked, needless, ms })
      }
      expect(true).toBe(true)
    }, 120000)
  }
})
