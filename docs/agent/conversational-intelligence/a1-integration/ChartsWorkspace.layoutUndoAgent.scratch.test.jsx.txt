// ── preferences: a store, so a write re-renders like the real hook ──
const prefStore = (() => {
  let prefs = {}
  const subs = new Set()
  return {
    get: () => prefs,
    set: (p) => { prefs = p; subs.forEach(f => f()) },
    write: (k, v) => { prefs = { ...prefs, [k]: v }; subs.forEach(f => f()) },
    subscribe: (f) => { subs.add(f); return () => subs.delete(f) },
  }
})()
const setPref = vi.fn((k, v) => { prefStore.write(k, v); return Promise.resolve(true) })
vi.mock('../../hooks/usePreferences', () => ({
  default: function usePreferences() { return { prefs: useSyncExternalStore(prefStore.subscribe, prefStore.get), setPref, loading: false } },
  parsePref: (raw, fallback) => {
    if (raw == null) return fallback
    if (typeof raw !== 'string') return raw
    try { return JSON.parse(raw) } catch { return fallback }
  },
}))

let modelCalls
beforeEach(() => {
  modelCalls = 0
  // jsdom's own storage (this file only): no conversation / in-flight marker from an earlier test
  try { localStorage.clear() } catch { /* */ }
  library = { global: [FIRM], mine: [MAIN, SWING, MACRO] }
  localStorage.setItem(AGENT_FLAG_KEY, '1')
  localStorage.setItem(AGENT_OPEN_KEY, '1')
  globalThis.fetch = vi.fn(async (url) => {
    const json = (o) => new Response(JSON.stringify(o), { status: 200, headers: { 'content-type': 'application/json' } })
    if (String(url).startsWith('/api/agent/turn')) { modelCalls += 1; return json({}) }
    if (String(url).startsWith('/api/agent/record')) return json({ conversationId: 'ac_1' })
    if (String(url).startsWith('/api/watchlists')) return json([])
    // the cross-tab guard's server read (host.boardInSync) is railed in its own suites; here it is
    // unreadable, which the guard treats as unchecked (never blocks on a network blip)
    if (String(url).startsWith('/api/auth/preferences')) return new Response('', { status: 503 })
    return json({ conversations: [] })
  })
})
afterEach(() => {
  cleanup()
  localStorage.removeItem(AGENT_FLAG_KEY)
  localStorage.removeItem(AGENT_OPEN_KEY)
  delete globalThis.fetch
})

async function say(text) {
  const box = await screen.findByLabelText('Message UCT Agent', {}, { timeout: 15000 })
  fireEvent.change(box, { target: { value: text } })
  await act(async () => { fireEvent.keyDown(box, { key: 'Enter' }) })
}

// SCRATCH (AGENT × INDICATORS A1.3b, 2026-10-13): the Agent's layout.open receipt vs the member's
// "Switched to …" toast — never both for one switch, never a stale or conflicting Undo.
const toastEl = () => screen.queryByTestId('layout-undo-toast')
const settle = (ms = 300) => act(async () => { await new Promise(r => setTimeout(r, ms)) })

test('L1 — an Agent open: the Agent receipt only, NO member toast', async () => {
  mount(MAIN)
  await say('open Swing Board')
  await waitFor(() => expect(activeId()).toBe(2), { timeout: 15000 })
  expect(await screen.findByText(/Opened “Swing Board” in place of “Main Trading”/, {}, { timeout: 15000 })).toBeTruthy()
  await settle()
  expect(toastEl()).toBeNull()
  expect(modelCalls).toBe(0)
}, 45000)

test('L2 — member Dock open (toast), THEN an Agent open: what does the member toast say, and its Undo?', async () => {
  mount(MAIN)
  await act(async () => { dockButton('Swing Board').click() })
  await waitFor(() => expect(activeId()).toBe(2))
  expect((await screen.findByTestId('layout-undo-toast')).textContent).toMatch(/^Switched to “Swing Board”/)
  await say('open Macro Board')
  await waitFor(() => expect(activeId()).toBe(3), { timeout: 15000 })
  expect(await screen.findByText(/Opened “Macro Board” in place of “Swing Board”/, {}, { timeout: 15000 })).toBeTruthy()
  await settle()
  expect(toastEl()).toBeNull()                                  // 1311765c84: the stale "Switched to Swing" is gone
  expect(modelCalls).toBe(0)
}, 45000)

test('L3 — Agent open, then member Dock open: Agent "undo" is refused (stale); the member toast Undo works', async () => {
  mount(MAIN)
  await say('open Swing Board')
  await waitFor(() => expect(activeId()).toBe(2), { timeout: 15000 })
  await screen.findByText(/Opened “Swing Board” in place of “Main Trading”/, {}, { timeout: 15000 })
  await act(async () => { dockButton('Macro Board').click() })
  await waitFor(() => expect(activeId()).toBe(3))
  expect((await screen.findByTestId('layout-undo-toast')).textContent).toMatch(/^Switched to “Macro Board”/)
  await say('undo')
  await settle(600)
  console.log('L3 OBSERVED Agent undo reply:', JSON.stringify(screen.getByTestId('agent-transcript').textContent.slice(-220)))
  expect(activeId()).toBe(3)                                     // the Agent never undoes the member's switch
  const t = toastEl()
  expect(t).toBeTruthy()
  await act(async () => { within(t).getByRole('button', { name: 'Undo' }).click() })
  await waitFor(() => expect(activeId()).toBe(2))
  expect(modelCalls).toBe(0)
}, 60000)
