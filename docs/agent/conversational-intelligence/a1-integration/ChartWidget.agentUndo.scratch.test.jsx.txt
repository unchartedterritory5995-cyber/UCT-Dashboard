import { peekIndicatorHistory } from '../../../components/chart/indicatorHistoryStore'
const historyKeys = () => { const out = []; for (let i = 0; i < sessionStorage.length; i++) { const k = sessionStorage.key(i); if (k && k.startsWith('uct.indicators.history.v1:')) out.push(k) } return out }

test('Agent add / hide / show / update / remove through the real ChartWidget: lands, records NOTHING', async () => {
  render(<Wrap initialOpts={{ tf: 'D' }} />)
  // StockChart is mocked here, so its toolbar never publishes 'can manage indicators'; only that gate is stubbed —
  // the write still goes through the REAL api.agent.commit → onOptsChange
  adapter().canManageIndicators = () => true
  const host = buildWorkspaceHost({
    chartApiById: api.chartApiById,
    getWidgets: () => [{ id: 'w1', type: 'chart', color: 'A', x: 0, y: 0, w: 24, h: 20 }],
    widgetLabel: (t) => t,
  })
  host.persist = async () => ({ ok: true })            // the board's server ACK (ChartsWorkspace persist)
  const run = async (action, args) => {
    adapter().canManageIndicators = () => true          // (re-applied: the widget re-registers its api on render)
    const plan = planOps(collectTargets(host, ['indicatorEdits']), [{ action, target: 'ixe:w1', args }], {}, { surface: 'charts' })
    expect(plan.ok, JSON.stringify(plan)).toBe(true)
    const prevAct = globalThis.IS_REACT_ACT_ENVIRONMENT
    globalThis.IS_REACT_ACT_ENVIRONMENT = false
    try { const res = await commitPlan(host, plan); expect(res.ok, JSON.stringify(res)).toBe(true); return res } finally { globalThis.IS_REACT_ACT_ENVIRONMENT = prevAct }
  }
  const insts = () => (api.opts.settings?.indicatorInstances || []).filter(i => i && !i.deleted)
  await run('indicator.add', { defId: 'rsi', inputs: null })
  const rsi = insts().find(i => i.defId === 'rsi')
  expect(rsi).toBeTruthy()
  await run('indicator.hide', { instance: rsi.instanceId })
  expect(insts().find(i => i.instanceId === rsi.instanceId).hidden).toBe(true)
  await run('indicator.show', { instance: rsi.instanceId })
  if (process.env.WITH_UPDATE) {
    await run('indicator.update', { instance: rsi.instanceId, inputs: [{ key: 'period', value: 7 }] })
    expect(insts().find(i => i.instanceId === rsi.instanceId).inputs.period).toBe(7)
  }
  await run('indicator.remove', { instance: rsi.instanceId })
  expect(insts().find(i => i.instanceId === rsi.instanceId)).toBeUndefined()
  for (const u of [null, 1, '1']) for (const k of ['chart:w1', 'global']) expect(peekIndicatorHistory(k, { userId: u }).past).toHaveLength(0)
  expect(historyKeys()).toEqual([])
}, 30000)
